  Widget _buildMobileBottomBar(BuildContext context) {
    return Container(
      padding: EdgeInsets.symmetric(horizontal: 24, vertical: 16),
      decoration: BoxDecoration(
        color: context.surfaceColor,
        border: Border(top: BorderSide(color: context.borderColor)),
      ),
      child: SafeArea(
        child: Row(
          children: [
            Expanded(
              child: OutlinedButton(
                onPressed: () {
                  ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text("Proceeding to checkout...")));
                },
                style: OutlinedButton.styleFrom(
                  side: BorderSide(color: Colors.blue),
                  shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(24)),
                  padding: EdgeInsets.symmetric(vertical: 16),
                ),
                child: Text("Buy Now", style: TextStyle(color: Colors.blue, fontWeight: FontWeight.bold, fontSize: 16)),
              ),
            ),
            SizedBox(width: 16),
            Expanded(
              child: ElevatedButton(
                onPressed: () async {
                   try {
                     await CartService.addToCart(widget.product.id, quantity: 1);
                     CartScreen.refreshNotifier.value = !CartScreen.refreshNotifier.value;
                     ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text("Berhasil ditambahkan ke keranjang")));
                   } catch (e) {
                     ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text("Gagal menambahkan: $e")));
                   }
                },
                style: ElevatedButton.styleFrom(
                  backgroundColor: Colors.blue,
                  foregroundColor: Colors.white,
                  shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(24)),
                  padding: EdgeInsets.symmetric(vertical: 16),
                ),
                child: Text("Add to Cart", style: TextStyle(fontWeight: FontWeight.bold, fontSize: 16)),
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildMobileLayout(BuildContext context) {
    return SingleChildScrollView(
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          // Custom Floating App Bar & Image
          Stack(
            children: [
              GestureDetector(
                onTap: () {
                  if (widget.product.imageUrl.isNotEmpty) {
                    Navigator.push(
                      context,
                      MaterialPageRoute(
                        builder: (context) => Scaffold(
                          backgroundColor: Colors.black,
                          appBar: AppBar(
                            backgroundColor: Colors.black,
                            iconTheme: IconThemeData(color: Colors.white),
                          ),
                          body: Center(
                            child: InteractiveViewer(
                              child: Image.network(
                                'http://127.0.0.1:8000${widget.product.imageUrl}',
                                fit: BoxFit.contain,
                              ),
                            ),
                          ),
                        ),
                      ),
                    );
                  }
                },
                child: Container(
                  height: 400,
                  width: double.infinity,
                  color: context.surfaceHighlight,
                  child: widget.product.imageUrl.isNotEmpty
                      ? Image.network('http://127.0.0.1:8000${widget.product.imageUrl}', fit: BoxFit.cover,

              ),
              SafeArea(

                      _buildFloatingIcon(Icons.arrow_back, () => Navigator.pop(context)),

  Widget _buildFloatingIcon(IconData icon, VoidCallback onTap, {Color? color}) {
    return GestureDetector(
      onTap: onTap,
      child: Container(
        padding: EdgeInsets.all(10),
        decoration: BoxDecoration(
          color: context.surfaceColor,
          shape: BoxShape.circle,
          boxShadow: [BoxShadow(color: Colors.black12, blurRadius: 4, offset: Offset(0, 2))],
        ),
        child: Icon(icon, color: color ?? context.textColor, size: 20),
      ),
    );
  }
